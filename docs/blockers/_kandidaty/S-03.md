  ```
     91               ? String(loc[loc.length - 1])
     92               : null
     93           if (msg && field) return `${field}: ${msg}`
     94           if (msg) return msg
     95           return JSON.stringify(x)
     96         })
     97         .filter((p): p is string => Boolean(p))
     98       // FastAPI/Pydantic часто возвращают много строк — ограничим длину.
     99       return parts.join('; ').slice(0, 240)
  ```
