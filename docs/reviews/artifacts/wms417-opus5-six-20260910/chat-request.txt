223: def _raise_chat_error(exc: chat_service.ChatError) -> NoReturn:
224:     code = exc.code
225:     if code == "forbidden":
226:         raise HTTPException(status.HTTP_403_FORBIDDEN, detail=code)
227:     if code in {"empty_message", "empty_file", "title_required"}:
228:         raise HTTPException(status.HTTP_400_BAD_REQUEST, detail=code)
229:     if code in {
230:         "attachment_not_owned",
231:         "document_seller_mismatch",
232:         "bad_document_kind",
233:         "cross_tenant",
234:         "client_message_id_required",
235:         "client_message_id_too_long",
236:     }:
237:         raise HTTPException(status.HTTP_422_UNPROCESSABLE_CONTENT, detail=code)
238:     if code == "document_not_found":
239:         raise HTTPException(status.HTTP_404_NOT_FOUND, detail=code)
240:     if code == "message_files_too_large":
241:         raise HTTPException(status.HTTP_413_REQUEST_ENTITY_TOO_LARGE, detail=code)
242:     if code in {"deleted", "client_message_id_conflict"}:
243:         raise HTTPException(status.HTTP_409_CONFLICT, detail=code)
244:     raise HTTPException(status.HTTP_500_INTERNAL_SERVER_ERROR, detail=code)

252:     attachments: list[ChatAttachment],
253: ) -> MessageOut:
254:     result = _message_out(msg, attachments)
255:     author = await session.get(User, msg.author_user_id)
256:     if author is not None and author.tenant_id == user.tenant_id:
257:         result.author_label = author.email
