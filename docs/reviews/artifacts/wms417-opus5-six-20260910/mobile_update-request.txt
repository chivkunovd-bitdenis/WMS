618: def _raise_cancellation_http(exc: FbsCancellationError) -> None:
619:     detail = envelope_from_exc(exc)
620:     if exc.code == "order_not_found":
621:         raise HTTPException(status_code=status.HTTP_404_NOT_FOUND, detail=detail)
622:     if exc.code in ("order_not_cancellable", "marketplace_not_supported"):
623:         raise HTTPException(status_code=status.HTTP_409_CONFLICT, detail=detail)
624:     if exc.code in ("seller_not_found", "missing_marketplace_token", "ozon_not_connected"):
625:         raise HTTPException(status_code=status.HTTP_400_BAD_REQUEST, detail=detail)
626:     # Отказы Ozon — не наша внутренняя ошибка, и отдавать по ним 500 значит
627:     # спрятать от оператора причину. «Причина отмены недоступна» — это конфликт
628:     # состояния, «транспорт выключен» — временная недоступность, остальное
629:     # пришло от маркетплейса.
630:     if exc.code in (
631:         "ozon_cancel_reason_unavailable",
632:         "ozon_cancel_not_available",
633:         "ozon_cancel_reason_message_required",
634:     ):
635:         raise HTTPException(status_code=status.HTTP_409_CONFLICT, detail=detail)
636:     if exc.code == "ozon_live_cancel_blocked":
637:         raise HTTPException(status_code=status.HTTP_503_SERVICE_UNAVAILABLE, detail=detail)
638:     if exc.code.startswith("wb_") or exc.code.startswith("ozon_"):
639:         raise HTTPException(status_code=status.HTTP_502_BAD_GATEWAY, detail=detail)
640:     raise HTTPException(status_code=status.HTTP_500_INTERNAL_SERVER_ERROR, detail=detail)

1121:         )
1122:         .order_by(FbsSupply.planned_shipment_date.asc(), FbsSupply.name.asc())
1123:     )

270:         val workspace = state.value.workspace ?: return { error("Откройте поставку перед сканированием") }
271:         val order = oldestMatchingOrder(workspace.orders, value)

275:         // Retain this scan's identity if the response is lost; retry never manufactures another scan.
276:         val action: suspend () -> Unit = action@{
277:             if (!sourceResolved) {
278:                 val options = required { api.fbs().pickOptions(workspace.supply.id) }
279:                 if (openedSupplyId != workspace.supply.id) return@action
280:                 val choices = availablePickSources(options.firstOrNull { it.productId == body.productId }?.locations.orEmpty())
281:                 require(choices.isNotEmpty()) { "Для ${order?.product?.name} нет доступного места подбора" }

291:             if (result.kind == "product") refresh(workspace.supply.id)

108:         val request = Request.Builder().url(url.newBuilder().addQueryParameter("check", System.currentTimeMillis().toString()).build())
109:             .cacheControl(CacheControl.FORCE_NETWORK).build()
110:         client.newCall(request).execute().use { response ->
111:             if (!response.isSuccessful) throw IOException("Не удалось проверить обновление (HTTP ${response.code}). Повторите позже.")
112:             val body = response.body ?: throw IOException("Сервер обновлений вернул пустой ответ.")

115:         }

128:             if (!response.isSuccessful) throw IOException("Не удалось скачать обновление (HTTP ${response.code}). Повторите позже.")
129:             val body = response.body ?: throw IOException("Сервер вернул пустой файл.")
130:             require(body.contentLength() == -1L || body.contentLength() == update.sizeBytes) { "Размер файла обновления не совпадает с заявленным." }
131:             body.byteStream().use { saveApk(it, target, update, progress) }
132:         }
133:         try { verify(target, update) } catch (error: Exception) { target.delete(); throw error }
134:         target
135:     }

68:      * При успехе сохраняет в currentSession и только тогда — base URL.

71:         baseUrl: String,
72:         email: String,
73:         password: String,
74:     ): Result<AuthSession> = withContext(Dispatchers.IO) {
75:         val trimmedEmail = email.trim()
76:         val normalizedUrl = validateAndNormalizeBaseUrl(baseUrl, allowCleartext = BuildConfig.DEBUG)
77:             .getOrElse { throwable ->
78:                 val failure = (throwable as? ru.wms.tsd.core.api.BaseUrlValidationException)
79:                     ?.toNetworkFailure()
80:                     ?: NetworkFailure.InvalidUrl
81:                 return@withContext Result.failure(failure)
82:             }
83: 
84:         try {
85:             val apiClient = apiClientFactory(normalizedUrl)
86:             val authApi = apiClient.createService(AuthApi::class.java)
87: 
88:             val response = authApi.loginRouteAuthLoginPost(LoginBody(trimmedEmail, password))
89:             if (!response.isSuccessful) {
90:                 return@withContext Result.failure(classifyHttpStatus(response.code(), forLogin = true))
91:             }
92: 
93:             val token = response.body()?.accessToken
94:                 ?: return@withContext Result.failure(NetworkFailure.InvalidResponse)

97:                 email = trimmedEmail,

104:                     authStore.saveStaff(it.copy(token = token))
105:                 }
106:             } catch (_: Exception) {
107:                 currentSession = null
108:                 return@withContext Result.failure(NetworkFailure.StorageFailure)
109:             }
110:             currentSession = session
111:             Result.success(session)

113:             throw error
114:         } catch (error: Exception) {

118:             )
119:         }

146:             ?: return Result.failure(AuthException.StaffNotFound())

154:         }

156:         val session = AuthSession(
157:             email = staff.email,
158:             displayName = staff.displayName,
159:             token = staff.token,
160:         )
161:         currentSession = session
162:         return Result.success(session)
163:     }
